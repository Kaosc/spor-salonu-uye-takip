// ----------------------------------------------------------------
// Üyeler
// ----------------------------------------------------------------

interface Member {
	uid: string // Firebase Auth UID. Kullanıcıyı için bir auth account oluşturulduğunda firebase tarafından otomatik olarak üretilir. Bu uid ile kullanıcıya ait tüm veriler ilişkilendirilir. Bu uid email'e bağlıdır ve değişmez. Yeni kullanıcı eklediğinizde random bir şifre ile kayıt edebilirsiniz. Daha sonra kullanıcı mobil uygulamadan şifresini sıfırlayabilir. Önce Auth hesabı oluşturulacak dönen uid ile member koleksiyonuna kullanıcı datası eklenecek. kullanıcıyı koleksiyona kaydederken bu uid'yi kullanın.
	email: string
	role: "MEMBER" // Her zaman MEMBER
	firstName: string
	lastName: string
	phoneNumber: string
	address?: string
	totalCheckIns?: number
	gender: "MALE" | "FEMALE" | "OTHER" | "UNSPECIFIED"
	birthDate?: Date | null
	bloodType?: string
	weight?: number | null // Kilo as KG
	height?: number | null // Boy as CM
	emergencyContact?: {
		name: string
		phone: string
	}
	isActive: boolean
	subscriptionStatus?: "ACTIVE" | "EXPIRED" | "CANCELLED" | "PAUSED" | "NONE"
	createdAt?: Date
	updatedAt?: Date
	createdBy?: string
}

// ----------------------------------------------------------------
// Dolaplar
// ----------------------------------------------------------------

interface Locker {
	id: number
	isOccupied: boolean
	occupiedByUid: string | null
	occupiedAt: Date | null
}

// ----------------------------------------------------------------
// Abonelikler random id ile koleksiyona eklenir kullanıcı idsi kullanılmaz. Çünkü bir kullanıcının birden fazla aboneliği olabilir.
//  Kullanıcıya ait abonelikleri çekerken döküman içindeki memberUid ile filtrelenir.
// ----------------------------------------------------------------

interface Subscription {
	id?: string
	memberUid: string // !! Member datasındaki uniq kullanıcı "uid" si
	firstName?: string
	lastName?: string
	packageType: "MONTHLY" | "QUARTERLY" | "YEARLY"
	startDate: Date
	endDate: Date
	price: number
	paymentMethod: "CASH" | "CREDIT_CARD" | "TRANSFER"
	status: "ACTIVE" | "EXPIRED" | "CANCELLED" | "PAUSED" | "NONE"
	pausedAt?: Date
	createdBy: string
	updatedBy?: string
	createdAt: Date
	updatedAt?: Date
	notes?: string
}

// ----------------------------------------------------------------
// Personel kullanıcılar. Admin ve Staff
// ----------------------------------------------------------------

interface User {
	uid: string // Personel uidsi
	email: string
	firstName: string
	lastName: string
	role: "ADMIN" | "STAFF"
	isActive: boolean
	createdAt: Date
	lastLoginAt?: Date
}

// ----------------------------------------------------------------
// Check-in ve check-out işlemleri için. "checkOutTime" kullanıcı checkout yapana kadar null olur.
// Checkout yapıldığında checkOutTime dolu olur ve lockerIdAtCheckout dolu olur
// ----------------------------------------------------------------

interface CheckIn {
	memberUid: string // !! Member datasındaki uniq kullanıcı "uid"'si
	firstName: string
	lastName: string
	checkInTime: Date | FieldValue
	checkOutTime: Date | FieldValue | null
	lockerIdAtCheckout: number | null
	lastCheckedInBy: string
	lastCheckedOutBy?: string
}
